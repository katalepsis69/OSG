# BTA OSG Public Intake Portal: System Crontab Entries
# Add this to the www-data crontab on the cloud VPS: crontab -u www-data -e
# The log directory must exist and be writable by www-data (oracle-setup.sh and
# the runbook create it; to do it by hand: sudo mkdir -p /var/log/osg-portal
# && sudo chown www-data:www-data /var/log/osg-portal). A redirect into
# /var/log itself aborts the whole job, because the cron shell cannot write there.

# Hourly batch milestone email notifications (runs every hour on the hour)
# Email links reuse the last public URL served by the web app (persisted automatically).
# To pin a fixed domain instead, set PORTAL_BASE_URL_PINNED in config.php, or
# uncomment and edit the line below:
# PORTAL_BASE_URL=https://portal.example.gov.ph
0 * * * * /usr/bin/php /var/www/osg-portal/cron/send_milestones.php >> /var/log/osg-portal/cron.log 2>&1
